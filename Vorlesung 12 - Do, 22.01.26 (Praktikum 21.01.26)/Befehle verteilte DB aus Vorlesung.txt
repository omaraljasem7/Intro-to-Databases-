 -- SERVER 1:
create table Kunden (
	id serial,
	vorname VARCHAR(50),
	nachname VARCHAR(50),
	email VARCHAR(80),
	strasse VARCHAR(80),
	plz VARCHAR(50),
	stadt VARCHAR(50),
	land VARCHAR(50)
) PARTITION by range (id);

-- JETZT KUNDEN EINFÜGEN: Fehler - keine Partitionen

CREATE EXTENSION postgres_fdw;

CREATE SERVER postgres_2 FOREIGN DATA WRAPPER postgres_fdw
    OPTIONS (dbname 'versandhandel', host 'postgres2.db', port 
    '5432');

CREATE USER MAPPING for postgres SERVER postgres_2 
    OPTIONS (user 'postgres', password 'sicherespasswort');
